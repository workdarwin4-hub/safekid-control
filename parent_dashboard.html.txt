        };
        firebase.initializeApp(firebaseConfig);
        const db = firebase.database();
        let currentParentId = null;

        function loginParent() {
            const id = document.getElementById('parentIdInput').value.trim();
            if (!id) return alert('برجاء إدخال Parent ID');

            db.ref('parents/' + id).once('value', snapshot => {
                const parent = snapshot.val();
                if (!parent) return alert('هذا الـ Parent ID غير موجود!');
                if (parent.status === 'banned') return alert('عذراً، هذا الحساب محظور بواسطة السوبر أدمن!');

                currentParentId = id;
                document.getElementById('loginModal').classList.add('hidden');
                document.getElementById('dashboard').classList.remove('hidden');
                document.getElementById('activeParentBadge').innerText = id;
                listenToDeviceCommands();
            });
        }

        function generatePairCode() {
            const code = Math.floor(100000 + Math.random() * 900000).toString();
            db.ref(`pairCodes/${code}`).set({
                parentId: currentParentId,
                status: 'pending',
                createdAt: Date.now()
            });
            document.getElementById('pairCodeDisplay').innerText = code;
        }

        function sendRemoteCommand(cmd) {
            if (!currentParentId) return;
            db.ref(`commands/${currentParentId}`).set({ command: cmd, timestamp: Date.now() });
            addLog(`تم إرسال أمر: ${cmd}`);
        }

        function listenToDeviceCommands() {
            db.ref(`deviceUpdates/${currentParentId}`).on('value', snapshot => {
                const data = snapshot.val();
                if (data && data.log) addLog(data.log);
            });
        }

        function addLog(msg) {
            const logs = document.getElementById('deviceLogs');
            logs.innerHTML += `<div>[${new Date().toLocaleTimeString('ar-EG')}] ${msg}</div>`;
            logs.scrollTop = logs.scrollHeight;
        }
